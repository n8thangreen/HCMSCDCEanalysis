model {
  # All stratifications
  # Loop over interventions/strategies
  for (i in 1:n_int) {

    # ============================================================
    # 1. MODELING STATE 1: ICD (Risks: Shock, Death)
    # ============================================================

    # Priors for RATES (Events per Person-Year)
    # Using uninformative Gamma(0.001, 0.001) priors
    rate[i, 1, 2] ~ dgamma(0.001, 0.001)  # Rate: ICD -> Shock
    rate[i, 1, 3] ~ dgamma(0.001, 0.001)  # Rate: ICD -> Death

    # Likelihood (Poisson count data)
    # IMPORTANT: 'n[i, 1]' must now be PERSON-YEARS, not just patient counts
    r[i, 1, 2] ~ dpois(rate[i, 1, 2] * n[i, 1])
    r[i, 1, 3] ~ dpois(rate[i, 1, 3] * n[i, 1])

    # Competing Risks Transformation
    # 1. Total rate of leaving state 1
    sum_rate_1[i] <- rate[i, 1, 2] + rate[i, 1, 3]

    # 2. Probability of ANY event happening in 1 cycle
    # (add 1E-10 to avoid division by zero if rates are 0)
    prob_event_1[i] <- 1 - exp(-sum_rate_1[i])

    # 3. Assign Probabilities to lambda matrix
    lambda[i, 1, 2] <- (rate[i, 1, 2] / (sum_rate_1[i] + 1.0E-10)) * prob_event_1[i]
    lambda[i, 1, 3] <- (rate[i, 1, 3] / (sum_rate_1[i] + 1.0E-10)) * prob_event_1[i]
    lambda[i, 1, 1] <- 1 - prob_event_1[i]  # Probability of staying
    
    # Zero out impossible transitions from state 1
    lambda[i, 1, 4] <- 0
    lambda[i, 1, 5] <- 0
    lambda[i, 1, 6] <- 0


    # ============================================================
    # 2. MODELING STATE 4: LOW RISK / NO ICD (Risks: SCD, Death)
    # ============================================================

    # Priors for RATES
    rate[i, 4, 5] ~ dgamma(0.001, 0.001)  # Rate: NoICD -> SCD
    rate[i, 4, 6] ~ dgamma(0.001, 0.001)  # Rate: NoICD -> Death

    # Likelihood
    r[i, 4, 5] ~ dpois(rate[i, 4, 5] * n[i, 4])
    r[i, 4, 6] ~ dpois(rate[i, 4, 6] * n[i, 4])

    # Competing Risks Transformation
    sum_rate_4[i] <- rate[i, 4, 5] + rate[i, 4, 6]
    prob_event_4[i] <- 1 - exp(-sum_rate_4[i])

    lambda[i, 4, 5] <- (rate[i, 4, 5] / (sum_rate_4[i] + 1.0E-10)) * prob_event_4[i]
    lambda[i, 4, 6] <- (rate[i, 4, 6] / (sum_rate_4[i] + 1.0E-10)) * prob_event_4[i]
    lambda[i, 4, 4] <- 1 - prob_event_4[i]  # Probability of staying
    
    # Zero out impossible transitions from state 4
    lambda[i, 4, 1] <- 0
    lambda[i, 4, 2] <- 0
    lambda[i, 4, 3] <- 0


    # ============================================================
    # 3. DETERMINISTIC & ABSORBENT STATES
    # ============================================================

    lambda[i, 2, 1:n_s] <- from_shock
    lambda[i, 3, 1:n_s] <- from_ICD_death

    # Absorbent states (SCD and Death)
    for (k in 5:n_s) {
      for (j in 1:n_s) {
        lambda[i, k, j] <- equals(k, j)
      }
    }
  }
}
