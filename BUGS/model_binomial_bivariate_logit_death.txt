model {
  # Bivariate normal priors on logit-probabilities of death to capture correlation

  # priors

  mu_death[1:2] ~ dmnorm(mean_prior[1:2], prec_prior[1:2, 1:2])
  Omega_death[1:2, 1:2] ~ dwish(R_wishart[1:2, 1:2], 2)
  Sigma_death[1:2, 1:2] <- inverse(Omega_death[1:2, 1:2])

  # for each scenario:
  # 1 = observed, 2 = >6% risk, 3 = >4% risk, 4 = no ICD
  for (i in 1:4) {
    # ICD death probability (col 3) and non-ICD death probability (col 6) are correlated
    logit_p_death[i, 1:2] ~ dmnorm(mu_death[1:2], Omega_death[1:2, 1:2])

    p_icd_death[i] <- ilogit(logit_p_death[i, 1])  # Prob: ICD -> Death
    p_noicd_death[i] <- ilogit(logit_p_death[i, 2])  # Prob: NoICD -> Death

    # Other probabilities are modeled independently using Beta priors
    p_icd_shock[i] ~ dbeta(1, 1)  # Prob: ICD -> Shock
    p_noicd_scd[i] ~ dbeta(1, 1)  # Prob: NoICD -> SCD

    # Binomial Likelihood for No ICD transitions (always has patients)
    r_noicd_scd[i] ~ dbin(p_noicd_scd[i], n_noicd[i])
    r_noicd_death[i] ~ dbin(p_noicd_death[i], n_noicd[i])

    # Track probabilities for diagnostics/plotting (keeping names matching downstream scripts)
    rate_icd_shock[i] <- p_icd_shock[i]
    rate_icd_death[i] <- p_icd_death[i]
    rate_noicd_scd[i] <- p_noicd_scd[i]
    rate_noicd_death[i] <- p_noicd_death[i]

    # transition probabilities for state 1 (ICD)
    lambda[i, 1, 1] <- 1 - p_icd_shock[i] - p_icd_death[i]
    lambda[i, 1, 2] <- p_icd_shock[i]
    lambda[i, 1, 3] <- p_icd_death[i]
    lambda[i, 1, 4] <- 0
    lambda[i, 1, 5] <- 0
    lambda[i, 1, 6] <- 0

    # transition probabilities for state 4 (No ICD)
    lambda[i, 4, 1] <- 0
    lambda[i, 4, 2] <- 0
    lambda[i, 4, 3] <- 0
    lambda[i, 4, 4] <- 1 - p_noicd_scd[i] - p_noicd_death[i]
    lambda[i, 4, 5] <- p_noicd_scd[i]
    lambda[i, 4, 6] <- p_noicd_death[i]

    # Deterministic / absorbent transitions
    lambda[i, 2, 1:6] <- from_shock
    lambda[i, 3, 1:6] <- from_ICD_death

    # Absorbent states
    for (k in 5:6) {
      for (j in 1:6) {
        lambda[i, k, j] <- equals(k, j)
      }
    }
  }

  # Binomial Likelihood for ICD transitions (only for scenarios 1, 2, 3 where n_icd > 0)
  for (i in 1:3) {
    r_icd_shock[i] ~ dbin(p_icd_shock[i], n_icd[i])
    r_icd_death[i] ~ dbin(p_icd_death[i], n_icd[i])
  }

  # Assign to separate strategy output matrices
  for (r in 1:6) {
    for (c in 1:6) {
      lambda.0[r, c] <- lambda[1, r, c]  # obs
      lambda.1[r, c] <- lambda[2, r, c]  # risk6
      lambda.2[r, c] <- lambda[3, r, c]  # risk4
      lambda.3[r, c] <- lambda[4, r, c]  # no_icd
    }
  }
}
