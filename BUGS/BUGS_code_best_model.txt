model {
  # Bivariate normal priors on log-rates of death to capture correlation
  mu_death[1:2] ~ dmnorm(mean_prior[1:2], prec_prior[1:2, 1:2])
  Omega_death[1:2, 1:2] ~ dwish(R_wishart[1:2, 1:2], 2)
  Sigma_death[1:2, 1:2] <- inverse(Omega_death[1:2, 1:2])

  for (i in 1:4) {
    # ICD death rate (col 3) and non-ICD death rate (col 6) are correlated
    log_rates_death[i, 1:2] ~ dmnorm(mu_death[1:2], Omega_death[1:2, 1:2])
    
    rate[i, 1, 3] <- exp(log_rates_death[i, 1])  # Rate: ICD -> Death
    rate[i, 4, 6] <- exp(log_rates_death[i, 2])  # Rate: NoICD -> Death

    # Other rates are modeled independently using Gamma priors
    rate[i, 1, 2] ~ dgamma(0.001, 0.001)  # Rate: ICD -> Shock
    rate[i, 4, 5] ~ dgamma(0.001, 0.001)  # Rate: NoICD -> SCD

    # Poisson Likelihood for observed transition counts
    r_icd_shock[i] ~ dpois(rate[i, 1, 2] * n_icd[i])
    r_icd_death[i] ~ dpois(rate[i, 1, 3] * n_icd[i])
    r_noicd_scd[i] ~ dpois(rate[i, 4, 5] * n_noicd[i])
    r_noicd_death[i] ~ dpois(rate[i, 4, 6] * n_noicd[i])

    # Competing Risks Transformation for State 1 (ICD)
    sum_rate_1[i] <- rate[i, 1, 2] + rate[i, 1, 3]
    prob_event_1[i] <- 1 - exp(-sum_rate_1[i])
    
    lambda[i, 1, 1] <- 1 - prob_event_1[i]
    lambda[i, 1, 2] <- (rate[i, 1, 2] / (sum_rate_1[i] + 1.0E-10)) * prob_event_1[i]
    lambda[i, 1, 3] <- (rate[i, 1, 3] / (sum_rate_1[i] + 1.0E-10)) * prob_event_1[i]
    lambda[i, 1, 4] <- 0
    lambda[i, 1, 5] <- 0
    lambda[i, 1, 6] <- 0

    # Competing Risks Transformation for State 4 (No ICD)
    sum_rate_4[i] <- rate[i, 4, 5] + rate[i, 4, 6]
    prob_event_4[i] <- 1 - exp(-sum_rate_4[i])
    
    lambda[i, 4, 1] <- 0
    lambda[i, 4, 2] <- 0
    lambda[i, 4, 3] <- 0
    lambda[i, 4, 4] <- 1 - prob_event_4[i]
    lambda[i, 4, 5] <- (rate[i, 4, 5] / (sum_rate_4[i] + 1.0E-10)) * prob_event_4[i]
    lambda[i, 4, 6] <- (rate[i, 4, 6] / (sum_rate_4[i] + 1.0E-10)) * prob_event_4[i]

    # Deterministic / absorbent transitions
    lambda[i, 2, 1:6] <- from_shock
    lambda[i, 3, 1:6] <- from_ICD_death

    for (k in 5:6) {
      for (j in 1:6) {
        lambda[i, k, j] <- equals(k, j)
      }
    }
  }

  # Assign to individual strategy output matrices
  for (r in 1:6) {
    for (c in 1:6) {
      lambda.0[r, c] <- lambda[1, r, c]
      lambda.1[r, c] <- lambda[2, r, c]
      lambda.2[r, c] <- lambda[3, r, c]
      lambda.3[r, c] <- lambda[4, r, c]
    }
  }
}
